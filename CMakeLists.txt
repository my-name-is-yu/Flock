cmake_minimum_required(VERSION 3.24)
project(flock VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
include(FetchContent)
find_package(SQLite3 REQUIRED)
find_package(CURL REQUIRED)
find_package(nlohmann_json 3.12 QUIET)
if(NOT nlohmann_json_FOUND)
  FetchContent_Declare(nlohmann_json
    URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.12.0
    URL_HASH SHA256=4b92eb0c06d10683f7447ce9406cb97cd4b453be18d7279320f7b2f025c10187)
  FetchContent_MakeAvailable(nlohmann_json)
endif()

add_executable(flock_workspace_supervisor src/workspace_supervisor.cpp)
target_compile_options(flock_workspace_supervisor PRIVATE -Wall -Wextra -Wpedantic)

add_library(flock_core src/store.cpp src/conversation.cpp src/conversation_agent.cpp src/conversation_actions.cpp
                       src/session_read.cpp src/model_budget.cpp src/runtime.cpp src/access.cpp src/access_tools.cpp src/secrets.cpp
                       src/bot_workspace.cpp src/bot_tools_external.cpp src/parallel_tools.cpp src/attachments.cpp)
target_include_directories(flock_core PUBLIC include)
if(TARGET SQLite3::SQLite3)
  target_link_libraries(flock_core PUBLIC SQLite3::SQLite3)
else()
  target_link_libraries(flock_core PUBLIC SQLite::SQLite3)
endif()
target_link_libraries(flock_core PUBLIC nlohmann_json::nlohmann_json PRIVATE CURL::libcurl)
add_dependencies(flock_core flock_workspace_supervisor)
target_compile_definitions(flock_core PRIVATE FLOCK_WORKSPACE_SUPERVISOR="$<TARGET_FILE:flock_workspace_supervisor>")
target_compile_options(flock_core PRIVATE -Wall -Wextra -Wpedantic)

include(CTest)
if(BUILD_TESTING)
  add_executable(flock_access_secrets tests/access_secrets.cpp)
  target_link_libraries(flock_access_secrets PRIVATE flock_core)
  add_test(NAME flock_access_secrets COMMAND flock_access_secrets)
  add_executable(flock_session_read tests/session_read.cpp)
  target_link_libraries(flock_session_read PRIVATE flock_core)
  target_compile_options(flock_session_read PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME flock_session_read COMMAND flock_session_read)
  add_executable(flock_core_tests tests/core.cpp)
  target_link_libraries(flock_core_tests PRIVATE flock_core)
  add_test(NAME flock_core COMMAND flock_core_tests)
  add_executable(flock_core_acceptance tests/core_acceptance.cpp)
  target_link_libraries(flock_core_acceptance PRIVATE flock_core)
  add_test(NAME flock_core_acceptance COMMAND flock_core_acceptance)
  add_executable(flock_conversation_store tests/conversation_store.cpp)
  target_link_libraries(flock_conversation_store PRIVATE flock_core)
  add_test(NAME flock_conversation_store COMMAND flock_conversation_store)
  add_executable(flock_conversation_runtime tests/conversation_runtime.cpp)
  target_link_libraries(flock_conversation_runtime PRIVATE flock_core)
  add_test(NAME flock_conversation_runtime COMMAND flock_conversation_runtime)
endif()

foreach(component t3 app web)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${component}/CMakeLists.txt")
    add_subdirectory(${component})
  endif()
endforeach()
if(TARGET flock AND TARGET flock_web)
  target_link_libraries(flock PRIVATE flock_web)
  target_compile_definitions(flock PRIVATE FLOCK_WITH_WEB=1)
endif()
if(TARGET flock AND TARGET flock_t3)
  target_link_libraries(flock PRIVATE flock_t3)
  target_compile_definitions(flock PRIVATE FLOCK_WITH_T3=1)
endif()
if(BUILD_TESTING AND TARGET flock_t3)
  add_executable(flock_live_t3 tests/live_t3.cpp)
  target_link_libraries(flock_live_t3 PRIVATE flock_t3)
endif()
if(BUILD_TESTING AND TARGET flock_t3 AND TARGET flock_app_support)
  add_executable(flock_live_evidence tests/live_evidence.cpp)
  target_include_directories(flock_live_evidence PRIVATE app)
  target_link_libraries(flock_live_evidence PRIVATE flock_t3 flock_app_support)
endif()
if(BUILD_TESTING AND TARGET flock_app_support)
  add_executable(flock_bot_tools_sandbox tests/bot_tools_sandbox.cpp)
  target_include_directories(flock_bot_tools_sandbox PRIVATE app)
  target_link_libraries(flock_bot_tools_sandbox PRIVATE flock_app_support)
  target_compile_options(flock_bot_tools_sandbox PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME flock_bot_tools_sandbox COMMAND flock_bot_tools_sandbox)
endif()
